def play_audio_file(file_path):
    """Windows Media Player OCX (COM) を利用して、MP3やWAVファイルをWindows上で非同期・同期再生する"""
    try:
        import win32com.client
        import time
        wmp = win32com.client.Dispatch("WMPlayer.OCX")
        wmp.settings.volume = 100
        wmp.URL = os.path.abspath(file_path)
        wmp.controls.play()
        
        # 再生開始を検知するまで極短時間待機
        time.sleep(0.3)
        
        # 再生中の状態（3: Playing, 6: Buffering, 9: Transitioning）の間ブロックする
        # これにより、連続再生などで同期（再生終了までブロック）を実現します
        while wmp.playState in [3, 6, 9]:
            time.sleep(0.1)
    except Exception as e:
        # fallback to winsound (WAVのみサポート、MP3の場合はWindowsの再生機能に依存)
        if winsound:
            try:
                winsound.PlaySound(file_path, winsound.SND_FILENAME)
            except Exception as ex:
                print(f"Fallback PlaySound error: {ex}")

    def restore_default_characters(self, parent_win):
        default_chars = config.load_default_characters()
        if not default_chars:
            messagebox.showwarning("エラー", "デフォルトのキャラクター設定が見つかりませんでした。", parent=parent_win)
            return
            
        restored_count = 0
        existing_names = [char["name"] for char in self.characters.values()]
        
        for d_key, d_data in default_chars.items():
            if d_data["name"] not in existing_names:
                new_key = d_key
                while new_key in self.characters:
                    new_key = new_key + "_"
                
                self.characters[new_key] = d_data
                self.char_keys.append(new_key)
                existing_names.append(d_data["name"])
                restored_count += 1
                
        if restored_count > 0:
            config.save_characters(self.characters)
            self.update_char_listbox()
            messagebox.showinfo("復旧完了", f"{restored_count}人のキャラクターを復旧しました！", parent=parent_win)
        else:
            messagebox.showinfo("復旧なし", "復旧できるキャラクターはありませんでした（すべて登録済みです）。", parent=parent_win)

        # キャンバスクリア
        self.repack_speech_rows()
        
        # スクロール位置を最下部に移動
        self.root.after(100, lambda: self.canvas.yview_moveto(1.0))
        return row_data

    def insert_speech_row_after(self, idx):
        """指定行 of list の直後に新しく行を挿入する"""
        row_frame = ttk.Frame(self.scrollable_frame, padding=5)
        
        current_char_key = self.speech_rows[idx]["char_key"] if idx < len(self.speech_rows) else "Zephyr"
        char_data = self.characters.get(current_char_key, {})
        emoji = char_data.get("emoji", "👤")
        display_name = char_data.get("name", current_char_key).split(" ")[0]
        
        avatar_btn = tk.Button(
            row_frame, 
            text=f"{emoji}\n{display_name}", 
            font=(self.font_family, 8), 
            width=6, 
            height=2,
            bg="#f5f5f5",
            relief="raised",
            bd=1
        )
        avatar_btn.pack(side=tk.LEFT, padx=(5, 10))
        
        text_widget = tk.Text(row_frame, font=(self.font_family, 11), height=2, wrap=tk.WORD, undo=True)
        text_widget.pack(side=tk.LEFT, fill=tk.X, expand=True)
        
        delete_btn = tk.Button(
            row_frame, 
            text="🗑️", 
            font=(self.font_family, 10), 
            fg="red", 
            bg="#fbf0f0",
            relief="flat",
            command=lambda: self.delete_speech_row(row_frame)
        )
        
        # キャッシュクリアボタン（ホバーで表示）
        clear_cache_btn = tk.Button(
            row_frame,
            text="🔄",
            font=(self.font_family, 10),
            fg="blue",
            bg="#f0f8ff",
            relief="flat",
            command=lambda: self.clear_speech_cache(row_frame)
        )
        
        row_data = {
            "row_id": uuid.uuid4().hex,
            "frame": row_frame,
            "char_key": current_char_key,
            "avatar_btn": avatar_btn,
            "text_widget": text_widget,
            "delete_btn": delete_btn,
            "clear_cache_btn": clear_cache_btn
        }
        
        self.speech_rows.insert(idx + 1, row_data)
        
        # 再配置と再バインド
        self.repack_speech_rows()
        self.rebind_all_rows()
        
        self.select_row(idx + 1)
        row_data["text_widget"].focus_set()
        return "break"

    def delete_speech_row(self, frame):
        """タイムラインから行を削除する（最低1行残す）"""
        if len(self.speech_rows) <= 1:
            messagebox.showwarning("警告", "これ以上行を削除できません（最低1行は必要です）")
            return
            
        target_idx = -1
        for idx, row in enumerate(self.speech_rows):
            if row["frame"] == frame:
                target_idx = idx
                break
                
        if target_idx != -1:
            row_data = self.speech_rows.pop(target_idx)
            row_data["frame"].destroy()
            
            self.repack_speech_rows()
            self.rebind_all_rows()
            
            new_idx = min(target_idx, len(self.speech_rows) - 1)
            self.select_row(new_idx)

    def repack_speech_rows(self):
        """全スピーチ行をCanvas上に再配置・描画する"""
        for row in self.speech_rows:
            row["frame"].pack_forget()
        for row in self.speech_rows:
            row["frame"].pack(fill=tk.X, expand=True, pady=2)
        self.canvas.configure(scrollregion=self.canvas.bbox("all"))

    def rebind_all_rows(self):
        """行数・並び順が変化した時にイベントバインドを補正する"""
        for idx, row in enumerate(self.speech_rows):
            # FocusIn再定義
            row["text_widget"].unbind("<FocusIn>")
            row["text_widget"].bind("<FocusIn>", lambda e, i=idx: self.select_row(i))
            
            # Return（Enter）再定義
            row["text_widget"].unbind("<Return>")
            row["text_widget"].bind("<Return>", lambda e, i=idx: self.insert_speech_row_after(i))
            
            # ホバーイベントの再定義
            self.setup_row_hover(row)
            
            row["avatar_btn"].config(command=lambda r=row: self.show_char_menu(r["avatar_btn"], r))

    def clear_speech_cache(self, row_frame):
        row_data = next((r for r in self.speech_rows if r["frame"] == row_frame), None)
        if not row_data:
            return
        row_id = row_data.get("row_id")
        if not row_id:
            return
            
        pattern = os.path.join(TEMP_DIR, f"speech_{row_id}_*.wav")
        files = glob.glob(pattern)
        deleted = 0
        for f in files:
            try:
                os.remove(f)
                deleted += 1
            except:
                pass
        
        if deleted > 0:
            messagebox.showinfo("キャッシュ削除", "この行の音声キャッシュを削除しました。\n次回再生時に再生成されます。")
        else:
            messagebox.showinfo("キャッシュ削除", "削除するキャッシュがありませんでした。")

    def setup_row_hover(self, row):
        """マウスホバーで行の右側にボタンを表示する処理"""
        frame = row["frame"]
        widget = row["text_widget"]
        btn = row["delete_btn"]
        cache_btn = row.get("clear_cache_btn")
        
        def on_enter(e):
            if len(self.speech_rows) > 1:
                btn.place(relx=1.0, rely=0.5, anchor="e", x=-5)
            if cache_btn:
                cache_btn.place(relx=1.0, rely=0.5, anchor="e", x=-35)
                
        def on_leave(e):
            x, y = frame.winfo_pointerxy()
            rx1 = frame.winfo_rootx()
            ry1 = frame.winfo_rooty()
            rx2 = rx1 + frame.winfo_width()
            ry2 = ry1 + frame.winfo_height()
            if not (rx1 <= x <= rx2 and ry1 <= y <= ry2):
                btn.place_forget()
                if cache_btn:
                    cache_btn.place_forget()
                
        frame.bind("<Enter>", on_enter)
        frame.bind("<Leave>", on_leave)
        widget.bind("<Enter>", on_enter)
        widget.bind("<Leave>", on_leave)
        btn.bind("<Enter>", on_enter)
        btn.bind("<Leave>", on_leave)
        if cache_btn:
            cache_btn.bind("<Enter>", on_enter)
            cache_btn.bind("<Leave>", on_leave)

    def show_char_menu(self, btn, row_data):
        """話者ボタンが押された際にキャラ選択用のプルダウンポップアップメニューを出す"""
        menu = tk.Menu(self.root, tearoff=0)
        
        favorites = [k for k in self.char_keys if self.characters[k].get("is_favorite", False)]
        display_keys = favorites if favorites else self.char_keys
        
        for key in display_keys:
            char_data = self.characters[key]
            emoji = char_data.get("emoji", "👤")
            name = char_data.get("name", key)
            prefix = "⭐ " if char_data.get("is_favorite", False) else ""
            menu.add_command(
                label=f"{prefix}{emoji} {name}", 
                command=lambda k=key, rd=row_data: self.change_row_character(rd, k)
            )
        
        x = btn.winfo_rootx()
        y = btn.winfo_rooty() + btn.winfo_height()
        menu.post(x, y)

    def change_row_character(self, row_data, char_key):
        """行の話者キャラクターを変更する"""
        row_data["char_key"] = char_key
        char_data = self.characters.get(char_key, {})
        emoji = char_data.get("emoji", "👤")
        display_name = char_data.get("name", char_key).split(" ")[0]
        
        row_data["avatar_btn"].config(text=f"{emoji}\n{display_name}")
        
        # 現在選択中のアクティブ行なら性格設定なども瞬時に更新
        current_idx = self.speech_rows.index(row_data)
        if current_idx == self.active_row_idx:
            self.select_row(current_idx)

    def select_row(self, idx):
        """行を選択状態にし、ハイライトと性格設定、パラメータ設定、左側リストを連動表示する"""
        if idx >= len(self.speech_rows):
            return
        
        self.active_row_idx = idx
        
        # ハイライト
        for i, row in enumerate(self.speech_rows):
            if i == idx:
                row["frame"].config(style="SelectedRow.TFrame")
                row["text_widget"].config(bg="#eef5fc", highlightbackground="#0078d4", highlightthickness=1)
            else:
                row["frame"].config(style="TFrame")
                row["text_widget"].config(bg="white", highlightbackground="#cccccc", highlightthickness=0)
        
        # キャラ固有設定の取得
        char_key = self.speech_rows[idx]["char_key"]
        char_data = self.characters.get(char_key, {})
        
        # フラグを立てて、スライダー連動時のTraceによる自動上書きセーブを防ぐ
        self.is_loading_character = True
        
        self.favorite_var.set(char_data.get("is_favorite", False))
        
        self.inst_text.delete("1.0", tk.END)
        self.inst_text.insert(tk.END, char_data.get("system_instruction", ""))
        self.inst_container.config(text=f"性格設定 (system_instruction) - 「{char_data.get('name', char_key)}」 - 自動保存されます")
        
        # パラメータスライダーの連動
        self.temp_var.set(char_data.get("temperature", 1.0))
        self.topp_var.set(char_data.get("top_p", 0.95))
        
        self.is_loading_character = False
        
        # 左側のキャラクター一覧の選択行も自動連動
        if hasattr(self, 'displayed_items') and char_key in self.displayed_items:
            char_idx = self.displayed_items.index(char_key)
            self.char_listbox.select_clear(0, tk.END)
            self.char_listbox.select_set(char_idx)
            self.char_listbox.see(char_idx)

    def on_favorite_change(self):
        if getattr(self, 'is_loading_character', False):
            return
        if not hasattr(self, 'speech_rows') or getattr(self, 'active_row_idx', -1) >= len(self.speech_rows):
            return
        
        char_key = self.speech_rows[self.active_row_idx]["char_key"]
        if char_key in self.characters:
            self.characters[char_key]["is_favorite"] = self.favorite_var.get()
            config.save_characters(self.characters)
            self.update_char_listbox()

    def delete_character(self):
        if not hasattr(self, 'speech_rows') or getattr(self, 'active_row_idx', -1) >= len(self.speech_rows):
            return
            
        char_key = self.speech_rows[self.active_row_idx]["char_key"]
        char_name = self.characters.get(char_key, {}).get("name", char_key)
        
        if len(self.char_keys) <= 1:
            messagebox.showwarning("警告", "最後の1人は削除できません。")
            return
            
        if not messagebox.askyesno("削除の確認", f"キャラクター「{char_name}」を削除しますか？\n（復旧する場合はヘルプ画面から行えます）"):
            return
            
        if char_key in self.char_keys:
            self.char_keys.remove(char_key)
        if char_key in self.characters:
            del self.characters[char_key]
            
        config.save_characters(self.characters)
        self.update_char_listbox()
        
        fallback_key = self.char_keys[0]
        for row in self.speech_rows:
            if row["char_key"] == char_key:
                self.change_row_character(row, fallback_key)
                
        messagebox.showinfo("削除完了", f"「{char_name}」を削除しました。")

    def on_parameter_change(self, *args):
        """スライダーの値を動的にキャラクターデータに保存する"""
        if self.is_loading_character:
            return
        idx = self.active_row_idx
        if idx < len(self.speech_rows):
            char_key = self.speech_rows[idx]["char_key"]
            if char_key in self.characters:
                self.characters[char_key]["temperature"] = round(self.temp_var.get(), 2)
                self.characters[char_key]["top_p"] = round(self.topp_var.get(), 2)
                config.save_characters(self.characters)

    def on_instruction_change(self, event):
        """性格設定テキストエリアが変更されたら即座にJSONに自動保存"""
        idx = self.active_row_idx
        if idx < len(self.speech_rows):
            char_key = self.speech_rows[idx]["char_key"]
            if char_key in self.characters:
                current_inst = self.inst_text.get("1.0", tk.END).strip()
                self.characters[char_key]["system_instruction"] = current_inst
                config.save_characters(self.characters)

    def update_char_listbox(self):
        self.char_listbox.delete(0, tk.END)
        self.displayed_items = []
        
        favorites = []
        others = []
        for key in self.char_keys:
            if self.characters[key].get("is_favorite", False):
                favorites.append(key)
            else:
                others.append(key)
                
        for key in favorites:
            name = self.characters[key]["name"]
            self.char_listbox.insert(tk.END, f"⭐ {name}")
            self.displayed_items.append(key)
            
        if favorites and others:
            self.char_listbox.insert(tk.END, "---")
            self.char_listbox.itemconfig(tk.END, {'fg': 'gray'})
            self.displayed_items.append(None)
            
        for key in others:
            name = self.characters[key]["name"]
            self.char_listbox.insert(tk.END, name)
            self.displayed_items.append(key)
            
        # Reselect active character if exists
        if hasattr(self, 'active_row_idx') and self.active_row_idx < len(self.speech_rows):
            char_key = self.speech_rows[self.active_row_idx]["char_key"]
            if char_key in self.displayed_items:
                idx = self.displayed_items.index(char_key)
                self.char_listbox.select_set(idx)
                self.char_listbox.see(idx)

    def on_character_list_select(self, event):
        """左側キャラリストで選択した際に、現在のアクティブ行の話者をそれに変更する"""
        selection = self.char_listbox.curselection()
        if selection and self.speech_rows:
            idx = selection[0]
            if not hasattr(self, 'displayed_items'):
                self.displayed_items = self.char_keys
                
            key = self.displayed_items[idx]
            if key is None:
                self.char_listbox.selection_clear(idx)
                active_row = self.speech_rows[self.active_row_idx]
                prev_key = active_row["char_key"]
                if prev_key in self.displayed_items:
                    self.char_listbox.selection_set(self.displayed_items.index(prev_key))
                return

            active_row = self.speech_rows[self.active_row_idx]
            # 既にそのキャラが設定されていればループを避ける
            if active_row["char_key"] != key:
                self.change_row_character(active_row, key)

    def on_play_click(self):
        """選択されている現在のアクティブ行のみ再生"""
        idx = self.active_row_idx
        if idx >= len(self.speech_rows):
            return
        
        row_data = self.speech_rows[idx]
        text = row_data["text_widget"].get("1.0", tk.END).strip()
        if not text:
            messagebox.showwarning("警告", "再生するテキストがありません。")
            return

        api_key = config.get_api_key()
        if not api_key:
            messagebox.showerror("APIキー未設定", "api_key.txtに有効なGemini APIキーを設定してください。")
            return

        self.stop_requested = False
        self.set_ui_state(True, "処理中 (connection...)")
        thread = threading.Thread(target=self._play_single_row, args=(row_data, text), daemon=True)
        thread.start()

    def _play_single_row(self, row_data, text):
        try:
            row_id = row_data.get("row_id", "default")
            char_key = row_data["char_key"]
            text_hash = hashlib.md5(text.encode('utf-8')).hexdigest()
            output_path = os.path.join(TEMP_DIR, f"speech_{row_id}_{char_key}_{text_hash}.wav")
            
            if os.path.exists(output_path):
                success = True
            else:
                success = self._api_synthesis_with_char(text, output_path, char_key)
                
            if success and not self.stop_requested:
                play_audio_file(output_path)
        except Exception as e:
            err_msg = str(e)
            # lambdaクロージャのエラーを回避するため実引数を直接afterに渡して表示
            self.root.after(0, self.show_error_dialog, "再生エラー", f"音声合成に失敗しました:\n{err_msg}")
        finally:
            self.root.after(0, self.set_ui_state, False, "")

    def on_play_all_click(self):
        """スピーチリストの全行を上の行から順番に「連続再生」"""
        api_key = config.get_api_key()
        if not api_key:
            messagebox.showerror("APIキー未設定", "api_key.txtに有効なGemini APIキーを設定してください。")
            return

        self.stop_requested = False
        self.set_ui_state(True, "処理中 (connection...)")
        thread = threading.Thread(target=self._play_all_rows, daemon=True)
        thread.start()

    def _play_all_rows(self):
        try:
            for idx, row in enumerate(self.speech_rows):
                if self.stop_requested:
                    break
                
                # 再生中の行を選択ハイライト
                self.root.after(0, lambda i=idx: self.select_row(i))
                
                text = row["text_widget"].get("1.0", tk.END).strip()
                if not text:
                    continue
                
                char_key = row["char_key"]
                row_id = row.get("row_id", f"idx_{idx}")
                text_hash = hashlib.md5(text.encode('utf-8')).hexdigest()
                output_path = os.path.join(TEMP_DIR, f"speech_{row_id}_{char_key}_{text_hash}.wav")
                
                if os.path.exists(output_path):
                    success = True
                else:
                    self.root.after(0, self.set_ui_state, True, f"処理中 (connection... {idx+1}/{len(self.speech_rows)})")
                    success = self._api_synthesis_with_char(text, output_path, char_key)
                    
                if success and not self.stop_requested:
                    # 同期再生（再生が完了するまで次の行に進まない）
                    play_audio_file(output_path)
        except Exception as e:
            err_msg = str(e)
            self.root.after(0, self.show_error_dialog, "連続再生エラー", f"エラーが発生しました:\n{err_msg}")
        finally:
            self.root.after(0, self.set_ui_state, False, "")

    def on_stop_click(self):
        """再生中の音声を停止する"""
        self.stop_requested = True
        try:
            import win32com.client
            wmp = win32com.client.Dispatch("WMPlayer.OCX")
            wmp.controls.stop()
        except:
            pass
        if winsound:
            try:
                winsound.PlaySound(None, winsound.SND_PURGE)
            except:
                pass

    def on_save_wav_click(self):
        """スピーチリスト全体を結合して1つのMP3に音声書き出し"""
        api_key = config.get_api_key()
        if not api_key:
            messagebox.showerror("APIキー未設定", "api_key.txtに有効なGemini APIキーを設定してください。")
            return

        file_path = filedialog.asksaveasfilename(defaultextension=".wav", filetypes=[("WAV files", "*.wav"), ("All files", "*.*")])
        if not file_path:
            return

        thread = threading.Thread(target=self._save_all_combined, args=(file_path,), daemon=True)
        thread.start()

    def _save_all_combined(self, file_path):
        try:
            temp_files = []
            for idx, row in enumerate(self.speech_rows):
                text = row["text_widget"].get("1.0", tk.END).strip()
                if not text:
                    continue
                    
                char_key = row["char_key"]
                row_id = row.get("row_id", f"idx_{idx}")
                text_hash = hashlib.md5(text.encode('utf-8')).hexdigest()
                temp_path = os.path.join(TEMP_DIR, f"speech_{row_id}_{char_key}_{text_hash}.wav")
                
                if os.path.exists(temp_path):
                    success = True
                else:
                    success = self._api_synthesis_with_char(text, temp_path, char_key)
                    
                if success:
                    temp_files.append(temp_path)

            if not temp_files:
                self.root.after(0, lambda: messagebox.showwarning("書き出し警告", "書き出すセリフがありません。"))
                return

            # WAVファイルの結合
            with wave.open(temp_files[0], 'rb') as first_w:
                nchannels = first_w.getnchannels()
                sampwidth = first_w.getsampwidth()
                framerate = first_w.getframerate()

            with wave.open(file_path, 'wb') as output_w:
                output_w.setnchannels(nchannels)
                output_w.setsampwidth(sampwidth)
                output_w.setframerate(framerate)
                for fp in temp_files:
                    with wave.open(fp, 'rb') as input_w:
                        output_w.writeframes(input_w.readframes(input_w.getnframes()))

            # 一時ファイル削除
            for fp in temp_files:
                try:
                    os.remove(fp)
                except:
                    pass

            self.root.after(0, self.show_info_dialog, "保存完了", f"音声を一括書き出ししました！\n{file_path}")
        except Exception as e:
            err_msg = str(e)
            self.root.after(0, self.show_error_dialog, "書き出しエラー", f"エラーが発生しました:\n{err_msg}")

    def _api_synthesis_with_char(self, text, output_file_path, char_key):
        from google import genai
        from google.genai import types

        api_key = config.get_api_key()
        client = genai.Client(api_key=api_key)

        char_data = self.characters[char_key]
        system_instruction = char_data.get("system_instruction", "")
        voice_name = char_data["voice_name"]
        temperature = char_data.get("temperature", 1.0)
        top_p = char_data.get("top_p", 0.95)

        async def _do_live_connect():
            config = types.LiveConnectConfig(
                response_modalities=["AUDIO"],
                system_instruction=types.Content(parts=[types.Part.from_text(text=system_instruction)]),
                speech_config=types.SpeechConfig(
                    voice_config=types.VoiceConfig(
                        prebuilt_voice_config=types.PrebuiltVoiceConfig(
                            voice_name=voice_name
                        )
                    )
                )
            )
            async with client.aio.live.connect(model="gemini-3.1-flash-live-preview", config=config) as session:
                await session.send(input=text, end_of_turn=True)
                
                self.root.after(0, self.set_ui_state, True, "処理中 (音声生成中)")
                
                audio_buffer = bytearray()
                import asyncio
                
                receive_iter = session.receive().__aiter__()
                while True:
                    try:
                        # 15秒間データが来なければ終了とみなす（タイムアウト）
                        response = await asyncio.wait_for(receive_iter.__anext__(), timeout=15.0)
                        
                        server_content = response.server_content
                        if server_content is not None:
                            model_turn = server_content.model_turn
                            if model_turn is not None:
                                for part in model_turn.parts:
                                    if part.inline_data:
                                        audio_buffer.extend(part.inline_data.data)
                            
                            # turn_completeが来たらループを抜ける
                            if server_content.turn_complete:
                                break
                    except asyncio.TimeoutError:
                        break
                    except StopAsyncIteration:
                        break
                    except Exception as e:
                        print(f"Receive error: {e}")
                        break
                
                # 明示的に接続を切断する
                if hasattr(session, 'close'):
                    try:
                        await session.close()
                    except Exception:
                        pass
                if audio_buffer:
                    with wave.open(output_file_path, "wb") as wf:
                        wf.setnchannels(1)
                        wf.setsampwidth(2)
                        wf.setframerate(24000)
                        wf.writeframes(audio_buffer)
                    return True
                return False

        if asyncio.run(_do_live_connect()):
            return True
        raise ValueError("音声データが応答から見つかりませんでした。")
